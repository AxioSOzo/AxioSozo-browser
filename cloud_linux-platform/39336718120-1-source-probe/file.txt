5://! The high-level module responsible for interfacing with the GPU.
6://!
7://! Much of WebRender's design is driven by separating work into different
8://! threads. To avoid the complexities of multi-threaded GPU access, we restrict
9://! all communication with the GPU to one thread, the render thread. But since
10://! issuing GPU commands is often a bottleneck, we move everything else (i.e.
11://! the computation of what commands to issue) to another thread, the
12://! RenderBackend thread. The RenderBackend, in turn, may delegate work to other
13://! thread (like the SceneBuilder threads or Rayon workers), but the
14://! Render-vs-RenderBackend distinction is the most important.
15://!
16://! The consumer is responsible for initializing the render thread before
17://! calling into WebRender, which means that this module also serves as the
18://! initial entry point into WebRender, and is responsible for spawning the
19://! various other threads discussed above. That said, WebRender initialization
20://! returns both the `Renderer` instance as well as a channel for communicating
21://! directly with the `RenderBackend`. Aside from a few high-level operations
22://! like 'render now', most of interesting commands from the consumer go over
23://! that channel and operate on the `RenderBackend`.
24://!
25://! ## Space conversion guidelines
26://! At this stage, we shuld be operating with `DevicePixel` and `FramebufferPixel` only.
27://! "Framebuffer" space represents the final destination of our rendeing,
28://! and it happens to be Y-flipped on OpenGL. The conversion is done as follows:
29://!   - for rasterized primitives, the orthographics projection transforms
30://! the content rectangle to -1 to 1
31://!   - the viewport transformation is setup to map the whole range to
32://! the framebuffer rectangle provided by the document view, stored in `DrawTarget`
33://!   - all the direct framebuffer operations, like blitting, reading pixels, and setting
34://! up the scissor, are accepting already transformed coordinates, which we can get by
35://! calling `DrawTarget::to_framebuffer_rect`

423:///
424:/// Manages the mapping between the at-a-distance texture handles used by the

427:struct TextureResolver {
428:    /// A map to resolve texture cache IDs to native textures.
429:    texture_cache_map: FastHashMap<CacheTextureId, CacheTexture>,
430:
431:    /// Map of external image IDs to native textures.
432:    external_images: FastHashMap<DeferredResolveIndex, ExternalTexture>,

434:    /// A special 1x1 dummy texture used for shaders that expect to work with
435:    /// the output of the previous pass but are actually running in the first
436:    /// pass.
437:    dummy_cache_texture: Texture,

440:impl TextureResolver {
441:    fn new(device: &mut Device) -> TextureResolver {
442:        let dummy_cache_texture = device
443:            .create_texture(
444:                ImageBufferKind::Texture2D,
445:                ImageFormat::RGBA8,
446:                1,
447:                1,
448:                TextureFilter::Linear,
449:                None,
450:            );

452:            &dummy_cache_texture,
453:            &[0xff, 0xff, 0xff, 0xff],
454:        );

456:        TextureResolver {
457:            texture_cache_map: FastHashMap::default(),
458:            external_images: FastHashMap::default(),
459:            dummy_cache_texture,
460:        }
461:    }
462:
463:    fn deinit(self, device: &mut Device) {
464:        device.delete_texture(self.dummy_cache_texture);
465:
466:        for (_id, item) in self.texture_cache_map {
467:            device.delete_texture(item.texture);
468:        }
469:    }
470:
471:    fn begin_frame(&mut self) {
472:    }
473:
474:    fn end_pass(
475:        &mut self,
476:        device: &mut Device,
477:        textures_to_invalidate: &[CacheTextureId],
478:    ) {
479:        // For any texture that is no longer needed, immediately
480:        // invalidate it so that tiled GPUs don't need to resolve it
481:        // back to memory.
482:        for texture_id in textures_to_invalidate {
483:            let render_target = &self.texture_cache_map[texture_id].texture;
484:            device.invalidate_render_target(render_target);
485:        }
486:    }
487:
488:    // Bind a source texture to the device.
489:    fn bind(&self, texture_id: &TextureSource, sampler: TextureSampler, device: &mut Device) -> Swizzle {
490:        match *texture_id {
491:            TextureSource::Invalid => {
492:                Swizzle::default()
493:            }
494:            TextureSource::Dummy => {
495:                let swizzle = Swizzle::default();
496:                device.bind_texture(sampler, &self.dummy_cache_texture, swizzle);
497:                swizzle
498:            }
499:            TextureSource::External(TextureSourceExternal { ref index, .. }) => {
500:                let texture = self.external_images
501:                    .get(index)
502:                    .expect("BUG: External image should be resolved by now");
503:                device.bind_external_texture(sampler, texture);
504:                Swizzle::default()
505:            }
506:            TextureSource::TextureCache(index, swizzle) => {
507:                let texture = &self.texture_cache_map[&index].texture;
508:                device.bind_texture(sampler, texture, swizzle);
509:                swizzle
510:            }
511:        }
512:    }

514:    // Get the real (OpenGL) texture ID for a given source texture.
515:    // For a texture cache texture, the IDs are stored in a vector
516:    // map for fast access.
517:    fn resolve(&self, texture_id: &TextureSource) -> Option<(&Texture, Swizzle)> {
518:        match *texture_id {
519:            TextureSource::Invalid => None,
520:            TextureSource::Dummy => {
521:                Some((&self.dummy_cache_texture, Swizzle::default()))
522:            }
523:            TextureSource::External(..) => {
524:                panic!("BUG: External textures cannot be resolved, they can only be bound.");
525:            }
526:            TextureSource::TextureCache(index, swizzle) => {
527:                Some((&self.texture_cache_map[&index].texture, swizzle))
528:            }
529:        }
530:    }

532:    // Retrieve the deferred / resolved UV rect if an external texture, otherwise
533:    // return the default supplied UV rect.

535:        &self,
536:        source: &TextureSource,
537:        default_value: TexelRect,
538:    ) -> TexelRect {
539:        match source {
540:            TextureSource::External(TextureSourceExternal { ref index, .. }) => {
541:                let texture = self.external_images
542:                    .get(index)
543:                    .expect("BUG: External image should be resolved by now");
544:                texture.get_uv_rect()
545:            }
546:            _ => {

548:            }

552:    /// Returns the size of the texture in pixels
553:    fn get_texture_size(&self, texture: &TextureSource) -> DeviceIntSize {
554:        match *texture {
555:            TextureSource::Invalid => DeviceIntSize::zero(),
556:            TextureSource::TextureCache(id, _) => {
557:                self.texture_cache_map[&id].texture.get_dimensions()
558:            },
559:            TextureSource::External(TextureSourceExternal { index, .. }) => {
560:                // If UV coords are normalized then this value will be incorrect. However, the
561:                // texture size is currently only used to set the uTextureSize uniform, so that
562:                // shaders without access to textureSize() can normalize unnormalized UVs. Which
563:                // means this is not a problem.
564:                let uv_rect = self.external_images[&index].get_uv_rect();
565:                (uv_rect.uv1 - uv_rect.uv0).abs().to_size().to_i32()
566:            },
567:            TextureSource::Dummy => DeviceIntSize::new(1, 1),
568:        }
569:    }

677:    /// (The number of frames since it was previously the backbuffer.)
678:    /// Returns an empty rect if the buffer is valid, and None if the entire buffer is invalid.
679:    fn get_damage_rect(&self, buffer_age: usize) -> Option<DeviceRect> {
680:        match buffer_age {
681:            // 0 means this is a new buffer, so is completely invalid.
682:            0 => None,
683:            // 1 means this backbuffer was also the previous frame's backbuffer
684:            // (so must have been copied to the frontbuffer). It is therefore entirely valid.
685:            1 => Some(DeviceRect::zero()),
686:            // We must calculate the union of the damage rects since this buffer was previously
687:            // the backbuffer.
688:            n if n <= self.damage_rects.len() + 1 => {
689:                Some(
690:                    self.damage_rects.iter()
691:                        .cycle()
692:                        .skip(self.current_offset + 1)
693:                        .take(n - 1)
694:                        .fold(DeviceRect::zero(), |acc, r| acc.union(r))
695:                )
696:            }
697:            // The backbuffer is older than the number of frames for which we track,
698:            // so we treat it as entirely invalid.
699:            _ => None,

1235:                        }

1326:    /// Renders the current frame.
1327:    ///
1328:    /// A Frame is supplied by calling [`generate_frame()`][webrender_api::Transaction::generate_frame].
1329:    /// buffer_age is the age of the current backbuffer. It is only relevant if partial present
1330:    /// is active, otherwise 0 should be passed here.
1331:    pub fn render(
1332:        &mut self,
1333:        device_size: DeviceIntSize,
1334:        buffer_age: usize,
1335:    ) -> Result<RenderResults, Vec<RendererError>> {
1336:        self.device_size = Some(device_size);
1337:
1338:        // TODO(gw): We want to make the active document that is
1339:        //           being rendered configurable via the public
1340:        //           API in future. For now, just select the last
1341:        //           added document as the active one to render
1342:        //           (Gecko only ever creates a single document
1343:        //           per renderer right now).
1344:        let doc_id = self.active_documents.keys().last().cloned();
1345:
1346:        let result = match doc_id {
1347:            Some(doc_id) => {
1348:                // Remove the doc from the map to appease the borrow checker
1349:                let mut doc = self.active_documents
1350:                    .remove(&doc_id)
1351:                    .unwrap();
1352:
1353:                let size = if !device_size.is_empty() {
1354:                    Some(device_size)
1355:                } else {
1356:                    None
1357:                };
1358:
1359:                #[cfg(feature = "debugger")]
1360:                let capture = self.renderdoc.take_request();
1361:                #[cfg(feature = "debugger")]
1362:                if capture {
1363:                    self.renderdoc.start();
1364:                }
1365:
1366:                let result = self.render_impl(
1367:                    doc_id,
1368:                    &mut doc,
1369:                    size,
1370:                    buffer_age,
1371:                );
1372:
1373:                #[cfg(feature = "debugger")]
1374:                if capture {
1375:                    let path = self.renderdoc.end();
1376:                    if let Some(reply) = self.renderdoc_capture_reply.take() {
1377:                        let result = match path {
1378:                            Some(p) => crate::api::debugger::RenderDocReply::Path(
1379:                                p.to_string_lossy().into_owned()
1380:                            ),
1381:                            None => crate::api::debugger::RenderDocReply::Error(
1382:                                "RenderDoc capture failed (launch the host with \
1383:                                 LD_PRELOAD=librenderdoc.so)".to_string()
1384:                            ),
1385:                        };
1386:                        let _ = reply.send(result);
1387:                    }
1388:                }
1389:
1390:                self.active_documents.insert(doc_id, doc);
1391:
1392:                result
1393:            }
1394:            None => {
1395:                self.last_time = zeitstempel::now();
1396:                Ok(RenderResults::default())
1397:            }
1398:        };
1399:
1400:        drain_filter(
1401:            &mut self.notifications,
1402:            |n| { n.when() == Checkpoint::FrameRendered },
1403:            |n| { n.notify(); },
1404:        );
1405:
1406:        let mut oom = false;
1407:        if let Err(ref errors) = result {
1408:            for error in errors {
1409:                if matches!(error, &RendererError::OutOfMemory) {
1410:                    oom = true;
1411:                    break;
1412:                }
1413:            }
1414:        }
1415:
1416:        if oom {
1417:            let _ = self.api_tx.send(ApiMsg::MemoryPressure);
1418:            // Ensure we don't get stuck in a loop.
1419:            self.consecutive_oom_frames += 1;
1420:            assert!(self.consecutive_oom_frames < 5, "Renderer out of memory");
1421:        } else {
1422:            self.consecutive_oom_frames = 0;
1423:        }
1424:
1425:        // This is the end of the rendering pipeline. If some notifications are is still there,
1426:        // just clear them and they will autimatically fire the Checkpoint::TransactionDropped
1427:        // event. Otherwise they would just pile up in this vector forever.
1428:        self.notifications.clear();
1429:
1430:        self.external_composite_debug_items = Vec::new();
1431:
1432:        tracy_rs::tracy_frame_marker!();
1433:
1434:        result
1435:    }

1933:                let old = self.texture_resolver.texture_cache_map.remove(&allocation.id);

1938:                    TextureCacheAllocationKind::Reset(_) |

1990:                    TextureCacheAllocationKind::Free => {}

2004:                                info.format,
2005:                                info.width,
2006:                                info.height,

2023:                            {
2024:                                self.clear_texture(&texture, [0.0; 4]);
2025:                            }

2027:                            // Textures in the cache generally don't need to be cleared,
2028:                            // but we do so if the debug display is active to make it
2029:                            // easier to identify unallocated regions.
2030:                            if self.debug_flags.contains(DebugFlags::TEXTURE_CACHE_DBG) {
2031:                                self.clear_texture(&texture, TEXTURE_CACHE_DBG_CLEAR_COLOR);
2032:                            }
2033:                        }
2034:
2035:                        create_cache_texture_time += zeitstempel::now() - create_cache_texture_start;
2036:

2038:                            texture,
2039:                            category: info.category,
2040:                        });
2041:                    }
2042:                    TextureCacheAllocationKind::Free => {}
2043:                };

2051:        if create_cache_texture_time > 0 {

2053:                profiler::CREATE_CACHE_TEXTURE_TIME,

2056:        }
2057:        if delete_cache_texture_time > 0 {
2058:            self.profile.set(
2059:                profiler::DELETE_CACHE_TEXTURE_TIME,
2060:                profiler::ns_to_ms(delete_cache_texture_time)
2061:            )
2062:        }
2063:
2064:        let t = self.profile.end_time(profiler::TEXTURE_CACHE_UPDATE_TIME);
2065:        self.resource_upload_time += t;

2067:
2068:        drain_filter(
2069:            &mut self.notifications,
2070:            |n| { n.when() == Checkpoint::FrameTexturesUpdated },
2071:            |n| { n.notify(); },
2072:        );
2073:    }

2105:    fn draw_instanced_batch<T: Clone>(
2106:        &mut self,
2107:        data: &[T],
2108:        vertex_array_kind: VertexArrayKind,
2109:        textures: &BatchTextures,
2110:        stats: &mut RendererStats,
2111:    ) {
2112:        if let Some(history) = &mut self.command_log {
2113:            history.draw(data.len() as u32);
2114:        }
2115:
2116:        self.bind_textures(textures);
2117:
2118:        // If we end up with an empty draw call here, that means we have
2119:        // probably introduced unnecessary batch breaks during frame
2120:        // building - so we should be catching this earlier and removing
2121:        // the batch.
2122:        debug_assert!(!data.is_empty());
2123:
2124:        let vao = &self.vaos[vertex_array_kind];
2125:        self.device.bind_vao(vao);
2126:

2128:            1

2130:            // Ensure each chunk will fit within the fixed size instance buffer.
2131:            vertex::SHARED_INSTANCE_BUFFER_SIZE / vao.instance_stride()

2133:            self.max_primitive_instance_count

2135:            data.len()
2136:        };

2138:        if self.use_shared_instance_buffer {
2139:            let instance_stride = vao.instance_stride();
2140:            for chunk in data.chunks(chunk_size) {
2141:                let offset = self
2142:                    .vaos
2143:                    .shared_instance_buffer
2144:                    .as_mut()
2145:                    .expect("shared instance buffer mode enabled but no shared buffer")
2146:                    .push_instances(&mut self.device, chunk);
2147:                let base_instance = (offset / instance_stride) as u32;
2148:                self.device.draw_indexed_triangles_instanced_base_instance_u16(
2149:                    6,
2150:                    chunk.len() as i32,
2151:                    base_instance,
2152:                );
2153:                self.profile.inc(profiler::DRAW_CALLS);
2154:                stats.total_draw_calls += 1;
2155:            }
2156:        } else {
2157:            for chunk in data.chunks(chunk_size) {
2158:                if self.enable_instancing {
2159:                    self.device
2160:                        .update_vao_instances(vao, chunk, ONE_TIME_USAGE_HINT, None);
2161:                    self.device
2162:                        .draw_indexed_triangles_instanced_u16(6, chunk.len() as i32);
2163:                } else {
2164:                    self.device
2165:                        .update_vao_instances(vao, chunk, ONE_TIME_USAGE_HINT, NonZeroUsize::new(4));
2166:                    self.device
2167:                        .draw_indexed_triangles(6 * chunk.len() as i32);
2168:                }
2169:                self.profile.inc(profiler::DRAW_CALLS);
2170:                stats.total_draw_calls += 1;
2171:            }
2172:        }
2173:
2174:        self.profile.add(profiler::VERTICES, 6 * data.len());
2175:    }

2177:    fn handle_readback_composite(
2178:        &mut self,
2179:        draw_target: DrawTarget,
2180:        uses_scissor: bool,
2181:        backdrop: &RenderTask,
2182:        readback: &RenderTask,
2183:    ) {
2184:        // Extract the rectangle in the backdrop surface's device space of where
2185:        // we need to read from.
2186:        let readback_origin = match readback.kind {
2187:            RenderTaskKind::Readback(ReadbackTask { readback_origin: Some(o), .. }) => o,
2188:            RenderTaskKind::Readback(ReadbackTask { readback_origin: None, .. }) => {
2189:                // If this is a dummy readback, just early out. We know that the
2190:                // clear of the target will ensure the task rect is already zero alpha,
2191:                // so it won't affect the rendering output.
2192:                return;
2193:            }
2194:            _ => unreachable!(),
2195:        };
2196:
2197:        if uses_scissor {
2198:            self.device.disable_scissor();
2199:        }
2200:
2201:        let texture_source = TextureSource::TextureCache(
2202:            readback.get_target_texture(),
2203:            Swizzle::default(),
2204:        );
2205:        let (cache_texture, _) = self.texture_resolver
2206:            .resolve(&texture_source).expect("bug: no source texture");
2207:
2208:        // Before submitting the composite batch, do the
2209:        // framebuffer readbacks that are needed for each
2210:        // composite operation in this batch.
2211:        let readback_rect = readback.get_target_rect();
2212:        let backdrop_rect = backdrop.get_target_rect();
2213:        let (backdrop_screen_origin, _) = match backdrop.kind {
2214:            RenderTaskKind::Picture(ref task_info) => (task_info.content_origin, task_info.device_pixel_scale),
2215:            _ => panic!("bug: composite on non-picture?"),
2216:        };
2217:
2218:        // Bind the FBO to blit the backdrop to.
2219:        // Called per-instance in case the FBO changes. The device will skip
2220:        // the GL call if the requested target is already bound.
2221:        let cache_draw_target = DrawTarget::from_texture(
2222:            cache_texture,
2223:            false,
2224:        );
2225:
2226:        // Get the rect that we ideally want, in space of the parent surface
2227:        let wanted_rect = DeviceRect::from_origin_and_size(
2228:            readback_origin,
2229:            readback_rect.size().to_f32(),
2230:        );
2231:
2232:        // Get the rect that is available on the parent surface. It may be smaller
2233:        // than desired because this is a picture cache tile covering only part of
2234:        // the wanted rect and/or because the parent surface was clipped.
2235:        let avail_rect = DeviceRect::from_origin_and_size(
2236:            backdrop_screen_origin,
2237:            backdrop_rect.size().to_f32(),
2238:        );
2239:
2240:        if let Some(int_rect) = wanted_rect.intersection(&avail_rect) {
2241:            // If there is a valid intersection, work out the correct origins and
2242:            // sizes of the copy rects, and do the blit.
2243:            let copy_size = int_rect.size().to_i32();
2244:
2245:            let src_origin = backdrop_rect.min.to_f32() +
2246:                int_rect.min.to_vector() -
2247:                backdrop_screen_origin.to_vector();
2248:
2249:            let src = DeviceIntRect::from_origin_and_size(
2250:                src_origin.to_i32(),
2251:                copy_size,
2252:            );
2253:
2254:            let dest_origin = readback_rect.min.to_f32() +
2255:                int_rect.min.to_vector() -
2256:                readback_origin.to_vector();
2257:
2258:            let dest = DeviceIntRect::from_origin_and_size(
2259:                dest_origin.to_i32(),
2260:                copy_size,
2261:            );
2262:
2263:            // Should always be drawing to picture cache tiles or off-screen surface!
2264:            debug_assert!(!draw_target.is_default());

2727:    fn handle_resolve(
2728:        &mut self,
2729:        resolve_op: &ResolveOp,
2730:        render_tasks: &RenderTaskGraph,
2731:        draw_target: DrawTarget,
2732:    ) {
2733:        for src_task_id in &resolve_op.src_task_ids {
2734:            let src_task = &render_tasks[*src_task_id];
2735:            let src_info = match src_task.kind {
2736:                RenderTaskKind::Picture(ref info) => info,
2737:                _ => panic!("bug: not a picture"),
2738:            };
2739:            let src_task_rect = src_task.get_target_rect().to_f32();
2740:
2741:            let dest_task = &render_tasks[resolve_op.dest_task_id];
2742:            let dest_info = match dest_task.kind {
2743:                RenderTaskKind::Picture(ref info) => info,
2744:                _ => panic!("bug: not a picture"),
2745:            };
2746:            let dest_task_rect = dest_task.get_target_rect().to_f32();
2747:
2748:            // If the dest picture is going to a blur target, it may have been
2749:            // expanded in size so that the downsampling passes don't introduce
2750:            // sampling error. In this case, we need to ensure we use the
2751:            // content size rather than the render task size to work out
2752:            // the intersecting rect to use for the resolve copy.
2753:            let dest_task_rect = DeviceRect::from_origin_and_size(
2754:                dest_task_rect.min,
2755:                dest_info.content_size.to_f32(),
2756:            );
2757:
2758:            // The rect we want to read, in the dest (resolve target) surface's
2759:            // raster space, normalized to a scale-independent space by dividing
2760:            // out the dest device pixel scale.
2761:            let wanted_rect_dest: WorldRect = DeviceRect::from_origin_and_size(
2762:                dest_info.content_origin,
2763:                dest_task_rect.size().to_f32(),
2764:            ).cast_unit() * dest_info.device_pixel_scale.inverse();
2765:
2766:            // Map it into the src (parent) surface's raster space. This is the
2767:            // identity unless the resolve target established a different raster
2768:            // root than the parent it reads back from (e.g. a backdrop-filter
2769:            // promoted to a root-snapping raster root inside a scrolled subtree),
2770:            // in which case it corrects for the offset between the two raster
2771:            // roots so we read back the region the backdrop actually covers.
2772:            let wanted_rect: WorldRect =
2773:                resolve_op.dest_to_src_raster.map_rect(&wanted_rect_dest);
2774:
2775:            // Get the rect that is available on the parent surface. It may be smaller
2776:            // than desired because this is a picture cache tile covering only part of
2777:            // the wanted rect and/or because the parent surface was clipped.
2778:            let avail_rect: WorldRect = DeviceRect::from_origin_and_size(
2779:                src_info.content_origin,
2780:                src_task_rect.size().to_f32(),
2781:            ).cast_unit() * src_info.device_pixel_scale.inverse();
2782:
2783:            // Both rects are now in the src surface's raster space, so the
2784:            // intersection is too.
2785:            if let Some(src_isect_rect) = wanted_rect.intersection(&avail_rect) {
2786:                let src_int_rect: DeviceRect =
2787:                    (src_isect_rect * src_info.device_pixel_scale).cast_unit();
2788:
2789:                // Map the intersection back into the dest surface's raster space
2790:                // to work out the region to write on the resolve target.
2791:                let dest_isect_rect: WorldRect =
2792:                    resolve_op.dest_to_src_raster.unmap_rect(&src_isect_rect);
2793:                let dest_int_rect: DeviceRect =
2794:                    (dest_isect_rect * dest_info.device_pixel_scale).cast_unit();
2795:
2796:                // If there is a valid intersection, work out the correct origins and
2797:                // sizes of the copy rects, and do the blit.
2798:
2799:                let src_origin = src_task_rect.min.to_f32() +
2800:                    src_int_rect.min.to_vector() -
2801:                    src_info.content_origin.to_vector();
2802:
2803:                let src = DeviceIntRect::from_origin_and_size(
2804:                    src_origin.to_i32(),
2805:                    src_int_rect.size().round().to_i32(),
2806:                );
2807:
2808:                let dest_origin = dest_task_rect.min.to_f32() +
2809:                    dest_int_rect.min.to_vector() -
2810:                    dest_info.content_origin.to_vector();
2811:
2812:                let dest = DeviceIntRect::from_origin_and_size(
2813:                    dest_origin.to_i32(),
2814:                    dest_int_rect.size().round().to_i32(),
2815:                );
2816:
2817:                let texture_source = TextureSource::TextureCache(
2818:                    src_task.get_target_texture(),
2819:                    Swizzle::default(),
2820:                );
2821:                let (cache_texture, _) = self.texture_resolver
2822:                    .resolve(&texture_source).expect("bug: no source texture");
2823:
2824:                let read_target = ReadTarget::from_texture(cache_texture);
2825:
2826:                // Should always be drawing to picture cache tiles or off-screen surface!
2827:                debug_assert!(!draw_target.is_default());
2828:                let device_to_framebuffer = Scale::new(1i32);
2829:
2830:                self.device.blit_render_target(
2831:                    read_target,
2832:                    src * device_to_framebuffer,
2833:                    draw_target,
2834:                    dest * device_to_framebuffer,
2835:                    TextureFilter::Linear,
2836:                );
2837:            }
2838:        }
2839:    }

3709:        device: &mut Device,
3710:        buffer: &GpuBuffer<T>,
3711:        dst_texture: &mut Option<Texture>,

3723:        }

3740:        let mut uploader = device.upload_texture(pbo_pool);
3741:

3746:                min: DeviceIntPoint::zero(),
3747:                max: DeviceIntPoint::new(buffer.size.width, buffer.size.height),

3749:            None,
